\begin{tikzpicture}[
    x=1cm,y=1cm,
    module/.style={draw=black!48,rounded corners=1.8mm,fill=white,
        minimum width=2.35cm,minimum height=1.05cm,align=center,
        font=\small,inner xsep=6pt,inner ysep=4pt},
    predictive/.style={module,draw=blue!55!black,fill=blue!4},
    ranking/.style={module,draw=orange!70!black,fill=orange!5},
    objective/.style={module,draw=violet!60!black,fill=violet!4},
    arrow/.style={-{Latex[length=2.2mm]},line width=.75pt,black!68},
    signal/.style={-{Latex[length=2.2mm]},line width=.8pt,green!42!black},
    gradient/.style={-{Latex[length=2.2mm]},line width=.8pt,violet!60!black},
    lane/.style={font=\small\bfseries,text=black!58,anchor=east}
]

% Quiet lane backgrounds keep inference and supervision visually separate.
\draw[rounded corners=2.5mm,fill=blue!1.5,draw=blue!10]
    (-0.25,1.1) rectangle (12.9,3.05);
\draw[rounded corners=2.5mm,fill=orange!1.5,draw=orange!12]
    (-0.25,-1.25) rectangle (12.9,0.72);
\node[lane] at (-0.42,2.08) {INFERENCE};
\node[lane] at (-0.42,-0.27) {TRAINING};

% Four balanced stages at inference.
\node[module] (candidate) at (1.25,2.08)
    {history $h_t$\\candidate phrase $a_i$};
\node[predictive] (world) at (4.35,2.08)
    {encode and predict\\$\hat z_i=F_\theta(\phi_\theta(h_t),e_\theta(a_i))$};
\node[ranking] (energy) at (7.65,2.08)
    {GAR energy\\$E_i=V_\psi(\hat z_i,z_x)$};
\node[module] (choice) at (11.15,2.08)
    {choose action\\$a_{\arg\min_i E_i}$};
\draw[arrow] (candidate) -- (world);
\draw[arrow] (world) -- (energy);
\draw[arrow] (energy) -- (choice);

% Observed counterfactual consequences yield two distinct training signals.
\node[module] (observations) at (1.25,-0.27)
    {observed successor set\\$\{T(h_t,a_i)\}_{i=1}^{K}$};
\node[predictive] (targets) at (4.35,-0.27)
    {EMA target states\\$\bar z_i,\ \bar z_g$};
\node[ranking] (labels) at (7.65,-0.27)
    {transition and order targets\\$\bar z_i,\ \bar d_i<\bar d_j$};
\node[objective] (losses) at (11.15,-0.27)
    {joint objective\\$\mathcal L_{\rm pred}+\mathcal L_{\rm rank}+\mathcal L_{\rm reg}$};
\draw[arrow] (observations) -- (targets);
\draw[signal] (targets) -- (labels);
\draw[gradient] (labels) -- (losses);

% The predictor supplies the successor used by both prediction and ranking.
\draw[gradient] (losses.north) -- ++(0,0.34) -| node[pos=.25,right,font=\scriptsize,text=violet!65!black]
    {update $\phi,e,F,V$} (world.south);
\draw[signal] (targets.north) -- ++(0,0.30) -| node[pos=.72,below,font=\scriptsize,text=green!42!black]
    {stop-gradient targets} (world.south west);
\end{tikzpicture}
